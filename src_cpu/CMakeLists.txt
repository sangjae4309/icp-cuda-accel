cmake_minimum_required(VERSION 3.18)
project(icp_cpu LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_path(EIGEN3_INCLUDE_DIR Eigen/Core PATH_SUFFIXES eigen3
          HINTS "$ENV{EIGEN3_INCLUDE_DIR}")
if(NOT EIGEN3_INCLUDE_DIR)
  message(FATAL_ERROR "Set -DEIGEN3_INCLUDE_DIR=/path/to/eigen3")
endif()
add_library(icp_core math.cpp io.cpp cpu_backend.cpp icp_run.cpp
  transform_points.cpp find_correspondences.cpp compute_statistics.cpp
  solve_increment.cpp update_pose.cpp)
target_include_directories(icp_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} PRIVATE ${EIGEN3_INCLUDE_DIR})
target_compile_options(icp_core PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-Wpedantic>)
if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU" AND CMAKE_CXX_COMPILER_VERSION VERSION_LESS 9)
  target_link_libraries(icp_core PUBLIC stdc++fs)
endif()
add_executable(icp_run main.cpp)
target_link_libraries(icp_run PRIVATE icp_core)
enable_testing()
add_executable(icp_tests tests/test_icp.cpp)
target_link_libraries(icp_tests PRIVATE icp_core)
add_test(NAME cpu_unit_tests COMMAND icp_tests)
